# Media Search Agent — Windows Native Python requirements

# Revisit this pin when upgrading opencv-python-headless to a NumPy-2-built wheel.

pymediainfo

# --- Embeddings & models ---
# torch and torchvision are installed by the Windows installers from the
# Blackwell-compatible CUDA index URL (cu128 or later — sm_120-supporting)
# before this file is processed. facenet-pytorch is installed by the shell
# installer with --no-deps after torch is in place (the legacy installer does
# not install facenet-pytorch). See header for the authoritative install order.
# `requests` is a runtime dep of facenet-pytorch (its inception_resnet_v1
# imports it at module load); needed here because --no-deps suppresses
# transitive resolution.
requests
# truststore bridges Python's stdlib SSL (used by torch.hub for the
# facenet-pytorch VGGFace2 download) to the OS trust store. Without it,
# fresh Windows VMs fail the facenet download with CERTIFICATE_VERIFY_FAILED
# because urllib doesn't read the Windows certificate store and doesn't
# load certifi by default. See src/msa_apps/search_api/setup_models.py.
truststore
open_clip_torch
# Object detection — RT-DETR backend (Apache-2.0, default)
# transformers <5: RT-DETR object detection (RTDetrForObjectDetection/RTDetrImageProcessor,
# added 4.45). Cap below 5 because the macOS bundle pins torch>=2.4,<2.7 (arm64 wheel
# stability) -> torch 2.6 / torchvision 0.21, which is too old for transformers 5.x's
# RT-DETR import (it needs newer torchvision); 4.x RT-DETR works on capped + uncapped torch.
transformers>=4.45,<5
tokenizers>=0.19.0   # transitive of transformers; pinned for stability
accelerate>=0.27.0   # transitive of transformers; needed for device_map
scenedetect
